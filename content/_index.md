        <script>
        (function() {
            if (document.getElementById('floating-letters-bg')) return;
            const bgContainer = document.createElement('div');
            bgContainer.id = 'floating-letters-bg';
            document.body.appendChild(bgContainer);
            
            const chars = ['A', 'B', 'C', 'X', 'Y', 'Z', 'α', 'β', 'γ', 'δ', 'ε', 'θ', 'λ', 'μ', 'π', 'σ', 'τ', 'φ', 'ω', '∑', '∫', '∆', 'Ω', '∞', '%', '$', '€', '£'];
            const colorVars = ['--floating-color-1', '--floating-color-2', '--floating-color-3', '--floating-color-4', '--floating-color-5'];
            
            function createFloatingLetter(delayMs = 0) {
                if (!document.body.contains(bgContainer)) return;
                
                const span = document.createElement('span');
                span.innerText = chars[Math.floor(Math.random() * chars.length)];
                
                const startX = Math.random() * 100; 
                const duration = 25 + Math.random() * 25; // 25s to 50s
                const fontSize = 1.5 + Math.random() * 5; 
                const colorVar = colorVars[Math.floor(Math.random() * colorVars.length)];
                
                // Static opacity per element (avoids CSS keyframe calculation flashes)
                const isDark = document.body.classList.contains('dark') || document.documentElement.getAttribute('data-theme') === 'dark';
                const minOp = isDark ? 0.05 : 0.15;
                const maxOp = isDark ? 0.18 : 0.45;
                const opacity = minOp + Math.random() * (maxOp - minOp);
                
                Object.assign(span.style, {
                    position: 'absolute',
                    left: `${startX}vw`,
                    top: `110vh`, // Start completely off-screen at the bottom
                    fontSize: `${fontSize}rem`,
                    color: `var(${colorVar})`,
                    opacity: opacity,
                    fontWeight: 'bold',
                    userSelect: 'none',
                    fontFamily: 'monospace',
                    transition: `transform ${duration}s linear, top ${duration}s linear`
                });
                
                // When this specific letter finishes its journey, delete it and spawn a new one!
                span.addEventListener('transitionend', (e) => {
                    if (e.propertyName === 'top') {
                        if (bgContainer.contains(span)) span.remove();
                        createFloatingLetter(0); // Instantly replace it to keep a constant number of letters
                    }
                });
                
                bgContainer.appendChild(span);
                
                // Wait for the requested delay, plus 50ms for the browser to register the initial DOM state
                setTimeout(() => {
                    if (!document.body.contains(bgContainer)) return;
                    span.style.top = '-20vh'; // Move past the top of the screen
                    const driftX = (Math.random() - 0.5) * 400; 
                    const rot = (Math.random() - 0.5) * 360; 
                    span.style.transform = `translateX(${driftX}px) rotate(${rot}deg)`;
                }, delayMs + 50);
            }
            
            // Spawn 50 letters, staggered perfectly so the screen starts empty and slowly fills up
            for(let i=0; i<50; i++) {
                // Stagger them randomly over the first 50 seconds
                createFloatingLetter(Math.random() * 50000); 
            }
            
            // Listen for theme changes to update opacity of existing letters dynamically
            const observer = new MutationObserver(() => {
                const isDark = document.body.classList.contains('dark') || document.documentElement.getAttribute('data-theme') === 'dark';
                const minOp = isDark ? 0.05 : 0.15;
                const maxOp = isDark ? 0.18 : 0.45;
                const letters = bgContainer.querySelectorAll('span');
                letters.forEach(l => {
                    l.style.opacity = minOp + Math.random() * (maxOp - minOp);
                });
            });
            observer.observe(document.body, { attributes: true, attributeFilter: ['class'] });
            observer.observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });
            
        })();
        </script>
